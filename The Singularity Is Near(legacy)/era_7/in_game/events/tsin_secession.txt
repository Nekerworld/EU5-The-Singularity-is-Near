# A grand compromise is proposed
tsin_secession.3 = {
	type = country_event
	category = disaster_event
	title = tsin_secession.3.title
	desc = tsin_secession.3.desc
	outcome = neutral
	illustration_tags = {
		10 = interior
		10 = regular
	}

	trigger = {
		NOT = { has_variable = tsin_secession_war }
		NOT = { has_variable = tsin_secession_compromise_refused }
	}

	# Accept the compromise: the crisis ends, slavery stays
	option = {
		name = tsin_secession.3.a
		set_variable = tsin_secession_resolved
		change_variable = { name = tsin_abolitionism subtract = 30 }
		add_country_modifier = { modifier = tsin_rotten_compromise years = 20 }
		add_estate_satisfaction = { type = estate_type:nobles_estate value = estate_satisfaction_mild_bonus }
		ai_chance = {
			factor = 3
		}
	}

	# No more compromises with slavery
	option = {
		name = tsin_secession.3.b
		set_variable = tsin_secession_compromise_refused
		change_variable = { name = tsin_abolitionism add = 5 }
		ai_chance = {
			factor = 5
		}
	}
}

# Secession: the slave regions leave the republic
tsin_secession.10 = {
	type = country_event
	category = disaster_event
	title = tsin_secession.10.title
	desc = tsin_secession.10.desc
	outcome = neutral
	image = "gfx/interface/illustrations/disaster/tsin_secession_crisis.dds"

	trigger = {
		NOT = { has_variable = tsin_secession_war }
		NOT = { has_variable = tsin_secession_resolved }
	}

	# Preserve the union by force
	option = {
		name = tsin_secession.10.a
		set_variable = tsin_secession_war
		create_rebel = {
			category = estate
			estate = nobles_estate
			rebel_gov_type = republic
			name = tsin_secessionist_rebels
			save_scope_as = tsin_secessionists
		}
		hidden_effect = {
			every_owned_location = {
				limit = {
					tsin_slave_region_location_trigger = yes
					is_capital = no
				}
				change_control = control_mild_penalty
				every_pop = {
					limit = { owner = root }
					add_pop_satisfaction = pop_satisfaction_ultimate_penalty
					change_pop_allegiance = scope:tsin_secessionists
				}
			}
		}
		custom_tooltip = tsin_secession.10.a.tt
		ai_chance = {
			factor = 10
		}
	}

	# One last compromise, whatever it costs
	option = {
		name = tsin_secession.10.b
		set_variable = tsin_secession_resolved
		add_stability = stability_mild_penalty
		change_variable = { name = tsin_abolitionism subtract = 20 }
		add_country_modifier = { modifier = tsin_rotten_compromise years = 30 }
		ai_chance = {
			factor = 2
		}
	}
}
