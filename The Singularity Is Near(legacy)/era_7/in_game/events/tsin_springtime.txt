namespace = tsin_springtime

# Springtime of Nations situation events (situations/tsin_springtime_of_nations.txt).
# Not flavor events: they do not spend the yearly tsin_ budget.

# The barricades go up
tsin_springtime.1 = {
	type = country_event
	title = tsin_springtime.1.title
	desc = tsin_springtime.1.desc
	outcome = neutral
	illustration_tags = {
		10 = exterior
		10 = angry
	}

	# Grant a constitution
	option = {
		name = tsin_springtime.1.a
		set_variable = tsin_springtime_conceded
		change_variable = { name = tsin_liberalism subtract = 30 }
		change_variable = { name = tsin_nationalism subtract = 10 }
		add_country_modifier = { modifier = tsin_springtime_constitution years = modifier_duration_years_long mode = add_and_extend }
		add_estate_satisfaction = { type = estate_type:nobles_estate value = estate_satisfaction_mild_penalty }
		change_societal_value = {
			type = absolutism_vs_liberalism
			value = societal_value_minor_move_to_right
		}
		ai_chance = {
			factor = 5
			modifier = {
				factor = 2
				societal_value:absolutism_vs_liberalism > 0
			}
		}
	}

	# Restore order by force
	option = {
		name = tsin_springtime.1.b
		set_variable = tsin_springtime_repressed
		add_country_modifier = { modifier = tsin_springtime_reaction years = modifier_duration_years_normal mode = add_and_extend }
		add_stability = stability_weak_penalty
		add_estate_satisfaction = { type = estate_type:nobles_estate value = estate_satisfaction_weak_bonus }
		ai_chance = {
			factor = 5
			modifier = {
				factor = 0
				stability < 0
			}
		}
	}

	# Wait and see
	option = {
		name = tsin_springtime.1.c
		ai_chance = {
			factor = 3
		}
	}
}

# Barricades in the capital
tsin_springtime.2 = {
	type = country_event
	title = tsin_springtime.2.title
	desc = tsin_springtime.2.desc
	outcome = neutral
	illustration_tags = {
		10 = exterior
		10 = armed
	}

	trigger = {
		has_country_modifier = tsin_springtime_barricades
	}

	# Negotiate with the committees
	option = {
		name = tsin_springtime.2.a
		change_gold_effect = { scale = -2 }
		change_variable = { name = tsin_liberalism subtract = 5 }
		ai_chance = {
			factor = 5
			modifier = {
				factor = 0
				gold < tsin_gold_cost_2
			}
		}
	}

	# Send in the troops
	option = {
		name = tsin_springtime.2.b
		capital ?= { set_variable = { name = tsin_labor_unrest value = 3 years = 5 } }
		add_stability = stability_weak_penalty
		change_variable = { name = tsin_liberalism add = 5 }
		ai_chance = {
			factor = 5
		}
	}
}

# The spring is over
tsin_springtime.10 = {
	type = country_event
	title = tsin_springtime.10.title
	desc = tsin_springtime.10.desc
	outcome = neutral
	illustration_tags = {
		10 = exterior
		10 = regular
	}

	option = {
		name = tsin_springtime.10.a
		trigger = { has_variable = tsin_springtime_conceded }
		remove_variable = tsin_springtime_conceded
		change_variable = { name = tsin_liberalism subtract = 10 }
	}

	option = {
		name = tsin_springtime.10.b
		trigger = { has_variable = tsin_springtime_repressed }
		remove_variable = tsin_springtime_repressed
		change_variable = { name = tsin_liberalism add = 10 }
		change_variable = { name = tsin_socialism add = 5 }
	}

	option = {
		name = tsin_springtime.10.c
		trigger = {
			NOT = { has_variable = tsin_springtime_conceded }
			NOT = { has_variable = tsin_springtime_repressed }
		}
	}
}
