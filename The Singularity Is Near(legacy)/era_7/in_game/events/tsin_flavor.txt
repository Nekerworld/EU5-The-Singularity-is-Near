namespace = tsin_flavor

# Regional flavor (M6): opening pressure, emancipation of the serfs, modernizing edict. Once per country each.
# Conditions are country states, never tags. .10 and .20 are fired from tsin_flavor_yearly_pulse and spend the
# flavor budget; .30 is the first stage of the tsin_defensive_modernization situation (docs/modernization.md).

# Gunboats in the harbour
tsin_flavor.10 = {
	type = country_event
	title = tsin_flavor.10.title
	desc = tsin_flavor.10.desc
	outcome = neutral
	illustration_tags = {
		10 = exterior
		10 = armed
	}

	trigger = {
		tsin_flavor_budget_available_trigger = yes
		current_date > 1840.1.1
		NOT = { has_variable = tsin_flavor_gunboats_done }
		NOT = { tsin_has_factory_trigger = yes }
		tsin_world_exhibition_leader_trigger = no
		any_owned_location = {
			is_port = yes
			NOT = { location_rank = location_rank:rural_settlement }
		}
	}

	immediate = {
		tsin_spend_flavor_budget_effect = yes
		set_variable = tsin_flavor_gunboats_done
	}

	# Open the ports by agreement
	option = {
		name = tsin_flavor.10.a
		tsin_add_opening_pressure_effect = yes
		add_country_modifier = { modifier = tsin_flavor_opened_ports years = modifier_duration_years_long mode = add_and_extend }
		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_weak_bonus }
		ai_chance = {
			factor = 10
		}
	}

	# Keep the ports closed
	option = {
		name = tsin_flavor.10.b
		add_prestige = prestige_weak_bonus
		add_estate_satisfaction = { type = estate_type:nobles_estate value = estate_satisfaction_weak_bonus }
		change_variable = { name = tsin_nationalism add = 5 }
		ai_chance = {
			factor = 10
		}
	}
}

# Emancipation of the serfs
tsin_flavor.20 = {
	type = country_event
	title = tsin_flavor.20.title
	desc = tsin_flavor.20.desc
	outcome = neutral
	illustration_tags = {
		10 = exterior
		10 = regular
	}

	trigger = {
		tsin_flavor_budget_available_trigger = yes
		current_date > 1845.1.1
		NOT = { has_variable = tsin_flavor_emancipation_done }
		OR = {
			has_estate_privilege = estate_privilege:noble_serfdom_rights
			societal_value:serfdom_vs_free_subjects < -20
		}
	}

	immediate = {
		tsin_spend_flavor_budget_effect = yes
		set_variable = tsin_flavor_emancipation_done
	}

	# Free them with land, paid off in redemption dues (shares V3 abolition effect)
	option = {
		name = tsin_flavor.20.a
		tsin_serfdom_abolish_redeemed_effect = yes
		change_societal_value = {
			type = serfdom_vs_free_subjects
			value = societal_value_large_move_to_right
		}
		add_estate_satisfaction = { type = estate_type:nobles_estate value = estate_satisfaction_mild_penalty }
		add_estate_satisfaction = { type = estate_type:peasants_estate value = estate_satisfaction_mild_bonus }
		add_country_modifier = { modifier = tsin_flavor_land_redemption years = modifier_duration_years_long mode = add_and_extend }
		ai_chance = {
			factor = 10
		}
	}

	# Free them without land (shares V3 abolition effect)
	option = {
		name = tsin_flavor.20.b
		tsin_serfdom_abolish_landless_effect = yes
		change_societal_value = {
			type = serfdom_vs_free_subjects
			value = societal_value_move_to_right
		}
		add_estate_satisfaction = { type = estate_type:peasants_estate value = estate_satisfaction_weak_penalty }
		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_weak_bonus }
		add_country_modifier = { modifier = tsin_flavor_landless_freedmen years = modifier_duration_years_long mode = add_and_extend }
		ai_chance = {
			factor = 5
		}
	}

	# Keep the old order
	option = {
		name = tsin_flavor.20.c
		change_societal_value = {
			type = serfdom_vs_free_subjects
			value = societal_value_minor_move_to_left
		}
		add_estate_satisfaction = { type = estate_type:nobles_estate value = estate_satisfaction_weak_bonus }
		add_estate_satisfaction = { type = estate_type:peasants_estate value = estate_satisfaction_mild_penalty }
		change_variable = { name = tsin_liberalism add = 5 }
		ai_chance = {
			factor = 5
		}
	}
}

# A modernizing edict
tsin_flavor.30 = {
	type = country_event
	title = tsin_flavor.30.title
	desc = tsin_flavor.30.desc
	outcome = neutral
	illustration_tags = {
		10 = interior
		10 = regular
	}

	trigger = {
		has_variable = tsin_treaty_ports_opened
		NOT = { has_variable = tsin_flavor_reform_edict_done }
		NOT = { has_variable = tsin_modernization_refused }
		NOT = { has_variable = tsin_modernization_failed }
		NOT = { has_variable = tsin_modernized }
	}

	# Proclaim a sweeping reorganization of the state
	option = {
		name = tsin_flavor.30.a
		tsin_start_modernization_effect = yes
		add_stability = stability_weak_penalty
		add_estate_satisfaction = { type = estate_type:clergy_estate value = estate_satisfaction_mild_penalty }
		add_country_modifier = { modifier = tsin_flavor_reorganization_edict years = modifier_duration_years_long mode = add_and_extend }
		change_variable = { name = tsin_liberalism add = 5 }
		ai_chance = {
			factor = 3
			modifier = {
				factor = 0
				stability < 0
			}
		}
	}

	# Adopt foreign machines, keep our own institutions
	option = {
		name = tsin_flavor.30.b
		tsin_start_modernization_effect = yes
		add_estate_satisfaction = { type = estate_type:nobles_estate value = estate_satisfaction_weak_bonus }
		add_country_modifier = { modifier = tsin_flavor_self_strengthening years = modifier_duration_years_long mode = add_and_extend }
		ai_chance = {
			factor = 10
		}
	}

	# Reject the foreign ways
	option = {
		name = tsin_flavor.30.c
		tsin_refuse_modernization_effect = yes
		add_prestige = prestige_weak_bonus
		add_estate_satisfaction = { type = estate_type:clergy_estate value = estate_satisfaction_mild_bonus }
		change_variable = { name = tsin_nationalism add = 5 }
		ai_chance = {
			factor = 5
		}
	}
}
