# Order and health laws (V4). Both are decided by bills (docs/legislation.md): every policy carries
# tsin_policy_enactable_trigger. The first policy of each law is its default state.

# Policing. Unlocked by tsin_modern_police (unlock_law). Only public order and unrest are handled here; the
# right to meet and organise stays with tsin_assembly_law, and the tsin_modern_police advance itself only
# gives global_separatism -0.1, so the policies below add on top of it.

	law_category = administrative

	# Default: night watch and militia.
	tsin_police_night_watch_policy = {
		allow = {
			tsin_policy_enactable_trigger = yes
		}

			nobles_estate_target_satisfaction = tiny_permanent_target_satisfaction
		}
		estate_preferences = {
			nobles_estate
		}
		years = 2
	}

	tsin_police_city_police_policy = {
		allow = {
			tsin_policy_enactable_trigger = yes
		}

			global_separatism = -0.05
