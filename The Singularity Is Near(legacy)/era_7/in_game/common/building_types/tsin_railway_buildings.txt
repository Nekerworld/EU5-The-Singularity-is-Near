# Maintenance: coal 0.1 (0.2), tools 0.05 (0.15), tsin_engines 0.03 (0.18).
# Two levels (tsin_railway_station_max_level); the telegraph tier saves coal and engines.

tsin_railway_station = {
	audio_tier = 2
	is_foreign = no
	pop_type = laborers
	max_levels = tsin_railway_station_max_level

	employment_size = tiny_rural_peasant_enhance_employment

	category = infrastructure_category

	town = yes
	city = yes
	megalopolis = yes

	location_potential = {
		OR = {
			is_capital = yes
			AND = {
				exists = owner
				has_road_of_type_to = {
					target = owner.capital
					type = road_type:railroad

			}
		}
	}

	# remove_if moved to cached-free connection modifier; assets are retained.

	build_time = infrastructure_build_time

	unique_production_methods = {
		tsin_railway_station_maintenance = {
			icon_type = building_type

			tools = 0.05
			tsin_engines = 0.03
			category = building_maintenance
		}
		tsin_railway_station_telegraph = {
			icon_type = goods
			icon = tsin_electricity
			coal = 0.07
			tools = 0.05
			tsin_engines = 0.02
			tsin_electricity = 0.01
			category = building_maintenance
		}
	}

	# modifier moved to cached-free connection modifier; assets are retained.

	construction_demand = tsin_factory_construction
}
