# Situation participation triggers (country scope).

# Industrial great power: same rank test as vanilla cb_imperialism create_enabled, plus industrial capitalism.
tsin_scramble_participant_trigger = {
	is_subject = no
	has_embraced_institution = institution:tsin_industrial_capitalism
	OR = {
		country_rank = country_rank:rank_kingdom
		country_rank = country_rank:rank_empire
		is_emperor = yes
	}
}

# Springtime of Nations: a country whose liberal, nationalist or socialist movement is strong. root = country.
tsin_springtime_unrest_trigger = {
	current_age_or_later = { age = age_7_industrialization }
	OR = {
		AND = {
			has_variable = tsin_liberalism
			var:tsin_liberalism >= 50
		}
		AND = {
			has_variable = tsin_nationalism
			var:tsin_nationalism >= 50
		}
		AND = {
			has_variable = tsin_socialism
			var:tsin_socialism >= 40
		}
	}
}

# Country scope. Among the eight countries with the most researched advances (ties count in favour).
tsin_world_exhibition_leader_trigger = {
	NOT = {
		any_country = {
			count >= 8
			num_of_advances_researched > root.num_of_advances_researched
		}
	}
}
