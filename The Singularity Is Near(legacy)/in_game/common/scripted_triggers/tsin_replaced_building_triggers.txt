# Location-scope triggers that accept a vanilla building or its age VII upgrade (docs/replaced_buildings.md).

tsin_has_university_tier_trigger = {
	OR = {
		has_building_with_at_least_one_level = university
		has_building_with_at_least_one_level = tsin_research_university
	}
}
