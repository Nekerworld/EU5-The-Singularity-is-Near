monthly_country_pulse = { on_actions = { tsin_ops_quarterly_pulse tsin_ops_monthly_alert_pulse } }
tsin_ops_quarterly_pulse = {
    trigger = { is_ai = no NOT = { has_variable = tsin_ops_scan_timer } }
    effect = { tsin_ops_scan_effect = yes }
}
tsin_ops_monthly_alert_pulse = {
    trigger = { is_ai = no NOT = { has_variable = tsin_ops_alert_timer } }
    effect = {
        set_variable = { name = tsin_ops_alert_timer months = 1 }
        tsin_ops_alert_cache_effect = yes
        if = { limit = { NOT = { has_variable = tsin_ops_ready } } set_variable = { name = tsin_ops_ready value = 1 } }
    }
}
