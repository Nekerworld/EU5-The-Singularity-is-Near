# Mean over great-power capital markets of market price / default price.
tsin_obs_gp_steel_price_ratio = {
	value = 0
	every_country = {
		limit = {
			is_great_power = yes
			capital ?= { market ?= { always = yes } }
		}
		add = {
			value = "capital.market.market_price(goods:steel)"
			divide = "capital.market.default_price(goods:steel)"
		}
	}
	if = {
		limit = { tsin_obs_great_powers > 0 }
		divide = tsin_obs_great_powers
	}
}

# 0.7 Modern Times indicators (docs/balance_targets.md section 10).
# country scope: monthly food balance per pop of the capital's market.
tsin_obs_capital_food_per_pop = {
	value = 0
	if = {
		limit = { capital ?= { market ?= { market_population > 0 } } }
		value = capital.market.market_monthly_food_balance
		divide = capital.market.market_population
	}
}

tsin_obs_gp_food = {
	value = 0
	every_country = {
		limit = { is_great_power = yes }
		add = tsin_obs_capital_food_per_pop
	}
	if = {
		limit = { tsin_obs_great_powers > 0 }
		divide = tsin_obs_great_powers
	}
}
