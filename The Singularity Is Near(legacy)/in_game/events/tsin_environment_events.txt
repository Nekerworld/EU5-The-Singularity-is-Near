# Environment events (docs/environment_events.md). Fired from tsin_envx_yearly_pulse (on_action/tsin_on_actions.txt).
# Each claims the shared yearly popup budget in immediate; the trigger refuses a second popup in the same year.

tsin_envx.1 = {
	type = country_event
	title = tsin_envx.1.title
	desc = tsin_envx.1.desc
	outcome = negative
	illustration_tags = { 10 = interior 10 = regular }

	trigger = {
		tsin_env_popup_available_trigger = yes is_ai = no has_variable = tsin_water_petition_cooldown any_owned_location = { var:tsin_pollution_water >= 60 var:tsin_pollution_water_bad_years >= 3 NOT = { has_variable = tsin_water_emergency_services } }
	}

	immediate = {
		tsin_env_popup_claim_effect = yes
		random_owned_location = { limit = { var:tsin_pollution_water >= 60 var:tsin_pollution_water_bad_years >= 3 NOT = { has_variable = tsin_water_emergency_services } } save_scope_as = tsin_envx_site }
	}

	option = {
		name = tsin_envx.1.a
		trigger = { gold >= 300 }
		add_gold = -300
		scope:tsin_envx_site = { set_variable = { name = tsin_water_emergency_services years = 5 } }
		ai_chance = { factor = 3 }
	}

	option = {
		name = tsin_envx.1.b
		ai_chance = { factor = 1 }
	}
}
