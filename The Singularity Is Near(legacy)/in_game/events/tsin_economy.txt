# Factory accident
tsin_economy.10 = {
	type = country_event
	title = tsin_economy.10.title
	desc = tsin_economy.10.desc
	outcome = negative
	illustration_tags = {
		10 = exterior
		10 = angry
	}

	trigger = {
		tsin_flavor_budget_available_trigger = yes
		tsin_has_factory_trigger = yes
	}

	immediate = {
		tsin_spend_flavor_budget_effect = yes
		random_owned_location = {
			limit = { tsin_location_has_factory_trigger = yes }
			save_scope_as = tsin_factory_location
		}
	}

	# Compensate the families
	option = {
		name = tsin_economy.10.a
		change_gold_effect = { scale = -2 }
		add_estate_satisfaction = { type = estate_type:peasants_estate value = estate_satisfaction_weak_bonus }
		ai_chance = {
			factor = 5
			modifier = {
				factor = 0
				gold < tsin_gold_cost_2
			}
		}
	}

	# Impose safety rules on the works
	option = {
		name = tsin_economy.10.b
		scope:tsin_factory_location = {
			add_location_modifier = { modifier = tsin_factory_safety_rules years = modifier_duration_years_normal mode = add_and_extend }
		}
		ai_chance = {
			factor = 10
		}
	}

	# Keep the lines running
	option = {
		name = tsin_economy.10.c
		scope:tsin_factory_location = {
			add_location_modifier = { modifier = tsin_factory_accident_unrest years = modifier_duration_years_short mode = add_and_extend }
		}
		add_estate_satisfaction = { type = estate_type:peasants_estate value = estate_satisfaction_mild_penalty }
		ai_chance = {
			factor = 3
			modifier = {
				factor = 0
				stability < 0
			}
		}
	}
}

# Rubber boom: turns tropical jungle locations into rubber gathering (tsin_rubber has no starting locations)
tsin_economy.80 = {
	type = country_event
	title = tsin_economy.80.title
	desc = tsin_economy.80.desc
	outcome = neutral
	illustration_tags = {
		10 = exterior
		10 = regular
	}

	trigger = {
		tsin_flavor_budget_available_trigger = yes
		current_year >= 1870
		has_advance = tsin_industrial_capitalism_advance
		NOT = { has_variable = tsin_rubber_boom_done }
		any_owned_location = {
			tsin_potential_rubber_location_trigger = yes
		}
	}

	immediate = {
		tsin_spend_flavor_budget_effect = yes
		set_variable = tsin_rubber_boom_done
	}

	# Grant concessions to rubber companies
	option = {
		name = tsin_economy.80.a
		ordered_owned_location = {
			limit = { tsin_potential_rubber_location_trigger = yes }
			order_by = {
				value = relative_raw_material_price

				text = tsin_economy.80.convert_tt
