# Oil prospecting, modelled on the vanilla prospect_for_coal decision.
tsin_prospect_for_oil = {
	decision_category = economy_decisions
	image = "gfx/interface/icons/trade_goods/illustrations/icon_goods_tsin_oil.dds"

	potential = {
		current_age_or_later = { age = age_7_industrialization }
		has_advance = tsin_machine_tools_advance
		scripted_geography:tsin_potential_oil = {
			any_location_in_scripted_geography = {
				owner = root
				is_city = no
				NOT = { raw_material = goods:tsin_oil }
			}
		}
	}

	allow = {
		political_influence >= coal_prospecting_cost
		OR = {
			is_ai = no
			NOT = { any_owned_location = { raw_material = goods:tsin_oil } }
		}
	}

	ai_will_do = {
		value = 100
	}

	option = {
		name = tsin_prospect_for_oil.a
		ai_chance = { value = 100 }
		effect = {
			add_political_influence = {
				value = coal_prospecting_cost
				multiply = -1
			}
			scripted_geography:tsin_potential_oil = {
				ordered_location_in_scripted_geography = {
					limit = {
						owner = root
						is_city = no
						NOT = { raw_material = goods:tsin_oil }
					}
					order_by = {
						value = relative_raw_material_price
						multiply = -1
					}
					max = 1
					save_scope_as = tsin_oil_prospect_location
					custom_tooltip = {
						text = tsin_prospect_for_oil_effect_tt
						change_max_raw_material_workers = {
							value = rgo_workers
							subtract = 1
							min = 0
							multiply = -1
						}
						change_raw_material = goods:tsin_oil
					}
				}
			}
		}
	}
}
