# Country scope. The country emergency caps reseeding; infections spread in the native engine.
tsin_cholera_seed_native_outbreak = {
	if = {
		limit = {
			var:tsin_cholera_outbreaks < tsin_cholera_max_outbreaks
			any_owned_location = {
				tsin_cholera_risk_location_trigger = yes
				NOT = { disease_has_outbreak_here = disease:tsin_cholera_disease }
			}
		}
		change_variable = { name = tsin_cholera_outbreaks add = 1 }
		ordered_owned_location = {
			limit = {
				tsin_cholera_risk_location_trigger = yes
				NOT = { disease_has_outbreak_here = disease:tsin_cholera_disease }
			}
			order_by = population
			max = 1
			save_scope_as = tsin_cholera_city
			if = {
				limit = { disease_is_active = disease:tsin_cholera_disease }
				set_disease_presence = {
					disease_outbreak = disease:tsin_cholera_disease.original_outbreak
					value = 0.1
				}
			}
			else = {
				spawn_disease = { disease = disease:tsin_cholera_disease value = 0.1 }
			}
		}
	}
}
