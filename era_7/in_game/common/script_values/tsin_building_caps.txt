# Age VII building level caps (compare vanilla building_caps.txt). Location scope.

# Automation needs a market for its larger output: on top of the vanilla mills penalty
# (market_access + 0.25 below 0.75), low market access cuts the cap a second time.
tsin_automation_max_level = {
	value = mills_max_level
	if = {
		limit = { market_access < 0.75 }
		multiply = {
			desc = "TSIN_BUILDING_LEVEL_AUTOMATION_MARKET_ACCESS"
			value = market_access
			add = 0.25
		}
	}
	min = 1
}

# Mechanized farm: 1 level from tsin_steam_threshing_advance (the unlock), +1 each from intensive farming and nitrogen fixation.
tsin_mechanized_farm_max_level = {
	add = {
		desc = "BUILDING_LEVEL_BASE"
		value = 1
	}
	if = {
		limit = { owner ?= { has_advance = tsin_intensive_farming_advance } }
		add = {
			desc = "TSIN_BUILDING_LEVEL_INTENSIVE_FARMING"
			value = 1
		}
	}
	if = {
		limit = { owner ?= { has_advance = tsin_nitrogen_fixation_advance } }
		add = {
			desc = "TSIN_BUILDING_LEVEL_NITROGEN_FIXATION"
			value = 1
		}
	}
}
