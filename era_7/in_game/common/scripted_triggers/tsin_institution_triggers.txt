# "Plausible" spawn locations for the age VII institutions (location scope).
# No continent requirement: the industrial world is wherever the industry is.

tsin_industrial_capitalism_plausible_location = {
	has_institution = institution:industrialization
	OR = {
		has_building_with_at_least_one_level = textile_mill
		has_building_with_at_least_one_level = iron_mill
		has_building_with_at_least_one_level = steel_mill
		has_building_with_at_least_one_level = tsin_mechanized_textile_mill
		has_building_with_at_least_one_level = tsin_automated_iron_mill
	}
	num_pop_type:laborers > 20
	total_building_levels > 150
	owner = {
		societal_value:capital_economy_vs_traditional_economy < -40
	}
}

tsin_mass_society_plausible_location = {
	has_institution = institution:enlightenment
	OR = {
		tsin_is_city_or_large_city = yes
		tsin_is_metropolis_or_megalopolis = yes
	}
	average_location_literacy > 40
	num_pop_type:laborers > 30
}

tsin_electrification_plausible_location = {
	has_institution = institution:tsin_industrial_capitalism
	tsin_has_university_tier_trigger = yes
	average_location_literacy > 50
	total_building_levels > 200
	owner = {
		societal_value:traditionalist_vs_innovative > 30
	}
}
