# Who finances industry: foreign-owned buildings (num_foreign_buildings) and the owner's investment policy.
# Estate ownership of buildings has no script trigger, so it is not shown (docs/mapmode_framework.md).
tsin_industrial_investment = {
	map_color = {
		if = {
			limit = {
				has_owner = yes
				num_foreign_buildings > 0
				tsin_location_mill_levels > 0
			}
			value = define:NMapColors|MAP_COLOR_MAX
		}
		else_if = {
			limit = {
				has_owner = yes
				num_foreign_buildings > 0
			}
			value = define:NMapColors|MAP_COLOR_HIGH
		}
		else_if = {
			limit = {
				has_owner = yes
				tsin_location_mill_levels > 0
				owner = { has_policy = policy:tsin_state_industry_policy }
			}
			value = define:NMapColors|MAP_COLOR_MID
		}
		else_if = {
			limit = {
				has_owner = yes
				tsin_location_mill_levels > 0
			}
