# Civic life around the age VII upgrade buildings (docs/replaced_buildings.md). Vanilla events that look for the
# old university or naval base skip upgraded locations; these take their place and ask for the upgrade ids.
# Research university .10-.50, steam port .60-.80, waterworks .90-.100, prefecture .110, automated mill .120.
# Fired from tsin_civic_yearly_pulse and spend the yearly flavor budget.

# The Humboldtian reform: research and teaching under one roof
tsin_civic.10 = {
	type = country_event
	title = tsin_civic.10.title
	desc = tsin_civic.10.desc
	illustration_tags = {
		10 = interior
		10 = regular
	}

	trigger = {
		tsin_flavor_budget_available_trigger = yes
		NOT = { has_variable = tsin_civic_humboldt_done }
		any_owned_location = { has_building_with_at_least_one_level = tsin_research_university }
	}

	immediate = {
		tsin_spend_flavor_budget_effect = yes
		set_variable = tsin_civic_humboldt_done
	}

	# Freedom of teaching and learning
	option = {
		name = tsin_civic.10.a
		add_country_modifier = { modifier = tsin_humboldt_reform years = 20 }
		add_estate_satisfaction = { type = estate_type:clergy_estate value = estate_satisfaction_mild_penalty }
		change_societal_value = {
			type = traditionalist_vs_innovative
			value = societal_value_minor_move_to_right
		}
		ai_chance = {
			factor = 10
		}
	}

	# The ministry sets the curriculum
	option = {
		name = tsin_civic.10.b
		add_country_modifier = { modifier = tsin_state_curriculum years = 20 }
		ai_chance = {
			factor = 5
		}
	}
}
