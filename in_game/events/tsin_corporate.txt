# Company chairman (docs/corporations.md). A country with companies keeps one chairman: a burgher character saved
# in var:tsin_chairman. tsin_corporate.10 appoints a new one from tsin_stock_monthly_pulse when there is none or he
# has died; .20-.60 come from tsin_corporate_yearly_pulse and spend the yearly flavor budget.

# Stock report (tsin_stock_report_decision): stock prices, the national index with its 8-quarter mean, the

# A new chairman
tsin_corporate.10 = {
	type = country_event
	title = tsin_corporate.10.title
	desc = tsin_corporate.10.desc
	outcome = neutral
	illustration_tags = {
		10 = interior
		10 = regular
	}

	trigger = {
		tsin_company_count > 0
		tsin_has_chairman_trigger = no
	}

	immediate = {
		create_character = {
			culture = root.culture
			religion = root.religion
			min_age = 35
			max_age = 55
			adm = { 40 90 }
			dip = { 30 70 }
			mil = { 0 30 }
			estate = estate_type:burghers_estate
			save_scope_as = tsin_chairman
		}
		set_variable = { name = tsin_chairman value = scope:tsin_chairman }
	}

	# Confirm the board's choice
	option = {
		name = tsin_corporate.10.a
		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_mild_bonus }
		ai_chance = {
			factor = 10
		}
	}

	# Seat a state commissioner on the board
	option = {
		name = tsin_corporate.10.b
		add_stability = stability_weak_bonus
		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_mild_penalty }
		tsin_scale_stocks_effect = { FACTOR = 0.95 }
		ai_chance = {
			factor = 3
		}
	}
}
