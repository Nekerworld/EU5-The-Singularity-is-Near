namespace = tsin_system

# Hidden: save migration. Runs from the yearly pulse once age 7 is active and the
# global setup from on_new_age_global never ran (mod enabled on an existing save).
tsin_system.1 = {
	type = country_event
	title = empty_text
	desc = empty_text
	outcome = neutral
	hidden = yes

	trigger = {
		NOT = { has_global_variable = tsin_version }
	}

	immediate = {
		set_global_variable = { name = tsin_version value = 1 }
		set_global_variable = { name = tsin_migrated_save value = 1 }
	}
}

		name = tsin_system.2.a
	}
}
