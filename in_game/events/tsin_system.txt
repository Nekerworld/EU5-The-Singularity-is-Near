namespace = tsin_system

# Hidden: save migration. Runs from the yearly pulse once age 7 is active and the
# global setup from on_new_age_global never ran (mod enabled on an existing save).
tsin_system.1 = {
	type = country_event
	title = empty_text
	desc = empty_text
	outcome = neutral
	hidden = yes

	trigger = {
		NOT = { has_global_variable = tsin_version }
	}

	immediate = {
		set_global_variable = { name = tsin_version value = 3 }
		set_global_variable = { name = tsin_migrated_save value = 1 }
		tsin_init_world_pop_base_effect = yes
	}
}

# Hidden: version 1 -> 3 (0.7). Saves from 0.6 get the world population base back-computed
# from this year's target curve, and the new global state starts empty. Version 3: countries that issued
# the modernizing edict before tsin_defensive_modernization existed resume at stage 1.
tsin_system.3 = {
	type = country_event
	title = empty_text
	desc = empty_text
	outcome = neutral
	hidden = yes

	trigger = {
		has_global_variable = tsin_version
		global_var:tsin_version < 3
	}

	immediate = {
		set_global_variable = { name = tsin_version value = 3 }
		if = {
			limit = { NOT = { has_global_variable = tsin_world_pop_base } }
			tsin_init_world_pop_base_effect = yes
		}
		every_country = {
			limit = {
				has_variable = tsin_flavor_reform_edict_done
				NOT = { has_variable = tsin_modernization_stage }
				NOT = { has_variable = tsin_modernized }
				NOT = { has_variable = tsin_modernization_failed }
			}
			set_variable = { name = tsin_modernization_stage value = 1 }
		}
	}
}
