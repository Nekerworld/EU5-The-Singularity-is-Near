namespace = tsin_millenarian

# The Millenarian Rebellion (0.7, docs/mass_rebellion.md). Disaster events of tsin_millenarian_rebellion.

# A prophet proclaims the heavenly kingdom
tsin_millenarian.1 = {
	type = country_event
	category = disaster_event
	title = tsin_millenarian.1.title
	desc = tsin_millenarian.1.desc
	outcome = neutral
	image = "gfx/interface/illustrations/disaster/tsin_millenarian_rebellion.dds"

	immediate = {
		create_rebel = {
			category = estate
			estate = peasants_estate
			name = tsin_millenarian_rebels
			save_scope_as = tsin_millenarian_rebels_scope
		}
		ordered_province = {
			limit = {
				population > 0
				NOT = { any_location_in_province = { is_capital = yes } }
			}
			order_by = {
				value = population
				if = {
					limit = { province_satisfaction > 0.1 }
					divide = province_satisfaction
				}
				else = { divide = 0.1 }
			}
			max = 8
			hidden_effect = {
				every_location_in_province = {
					change_control = control_mild_penalty
					every_pop = {
						limit = { owner = root }
						add_pop_satisfaction = pop_satisfaction_ultimate_penalty
						change_pop_allegiance = scope:tsin_millenarian_rebels_scope
					}
				}
			}
		}
	}

	# Crush the heretics with the imperial army
	option = {
		name = tsin_millenarian.1.a
		custom_tooltip = tsin_millenarian.1.tt
		change_gold_effect = { scale = -3 }
		add_country_modifier = { modifier = tsin_suppression_campaign years = 5 }
		ai_chance = {
			factor = 5
		}
	}

	# Let the provincial gentry raise their own armies
	option = {
		name = tsin_millenarian.1.b
		custom_tooltip = tsin_millenarian.1.tt
		add_country_modifier = { modifier = tsin_gentry_armies years = 15 }
		add_estate_satisfaction = { type = estate_type:nobles_estate value = estate_satisfaction_mild_bonus }
		ai_chance = {
			factor = 5
		}
	}
}
