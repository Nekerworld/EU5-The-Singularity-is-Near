# The People's Charter: a mass petition for manhood suffrage
tsin_society.10 = {
	type = country_event
	title = tsin_society.10.title
	desc = tsin_society.10.desc
	illustration_tags = {
		10 = exterior
		10 = angry
	}

	trigger = {
		tsin_flavor_budget_available_trigger = yes
		has_parliament = yes
		has_variable = tsin_liberalism
		var:tsin_liberalism >= 40
		NOT = { has_variable = tsin_society_charter_done }
	}

	immediate = {
		tsin_spend_flavor_budget_effect = yes
		set_variable = { name = tsin_society_charter_done value = yes years = 20 }
	}

	# Let parliament debate it
	option = {
		name = tsin_society.10.a
		change_variable = { name = tsin_liberalism subtract = 10 }
		add_estate_satisfaction = { type = estate_type:nobles_estate value = estate_satisfaction_mild_penalty }
		change_societal_value = {
			type = absolutism_vs_liberalism
			value = societal_value_minor_move_to_right
		}
		ai_chance = {
			factor = 5
		}
	}

	# Reject the petition
	option = {
		name = tsin_society.10.b
		change_variable = { name = tsin_liberalism add = 5 }
		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_mild_penalty }
		ai_chance = {
			factor = 5
		}
	}

	# Arrest the organizers
	option = {
		name = tsin_society.10.c
		change_variable = { name = tsin_liberalism subtract = 15 }
		change_variable = { name = tsin_socialism add = 5 }
		add_stability = stability_weak_penalty
		ai_chance = {
			factor = 3
			modifier = {
				factor = 0
				stability < 0
			}
		}
	}
}

# The strike
tsin_society.20 = {
	type = country_event
	title = tsin_society.20.title
	desc = tsin_society.20.desc
	illustration_tags = {
		10 = exterior
		10 = angry
	}

	trigger = {
		tsin_flavor_budget_available_trigger = yes
		has_variable = tsin_labor_unionism
		var:tsin_labor_unionism >= 40
		any_owned_location = { tsin_location_has_factory_trigger = yes }
	}

	immediate = {
		tsin_spend_flavor_budget_effect = yes
		random_owned_location = {
			limit = { tsin_location_has_factory_trigger = yes }
			save_scope_as = tsin_strike_location
		}
	}

	# Recognize the union
	option = {
		name = tsin_society.20.a
		change_variable = { name = tsin_labor_unionism subtract = 15 }
		add_estate_satisfaction = { type = estate_type:peasants_estate value = estate_satisfaction_weak_bonus }
		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_mild_penalty }
		ai_chance = {
			factor = 5
		}
	}

	# Break the strike
	option = {
		name = tsin_society.20.b
		add_country_modifier = { modifier = tsin_strike_broken years = modifier_duration_years_short mode = add_and_extend }
		change_variable = { name = tsin_labor_unionism add = 10 }
		change_variable = { name = tsin_socialism add = 5 }
		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_weak_bonus }
		ai_chance = {
			factor = 5
		}
	}

	# Arbitration
	option = {
		name = tsin_society.20.c
		change_gold_effect = { scale = -2 }
		change_variable = { name = tsin_labor_unionism subtract = 5 }
		ai_chance = {
			factor = 5
			modifier = {
				factor = 0
				gold < tsin_gold_cost_2
			}
		}
	}
}
