namespace = tsin_exhibition

# World exhibition (M5). .1 opens the exhibition for the host, .10 closes it.

# Opening: choose the centrepiece
tsin_exhibition.1 = {
	type = country_event
	title = tsin_exhibition.1.title
	desc = tsin_exhibition.1.desc
	illustration_tags = {
		10 = exterior
		10 = happy
	}

	# A hall of machines
	option = {
		name = tsin_exhibition.1.a
		add_country_modifier = { modifier = tsin_exhibition_machine_hall years = modifier_duration_years_long mode = add_and_extend }
		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_weak_bonus }
		ai_chance = {
			factor = 10
		}
	}

	# A palace of science and education
	option = {
		name = tsin_exhibition.1.b
		add_country_modifier = { modifier = tsin_exhibition_science_palace years = modifier_duration_years_long mode = add_and_extend }
		ai_chance = {
			factor = 5
		}
	}

	# National splendour
	option = {
		name = tsin_exhibition.1.c
		add_prestige = prestige_mild_bonus
		add_stability = stability_weak_bonus
		ai_chance = {
			factor = 5
		}
	}
}

# Closing
tsin_exhibition.10 = {
	type = country_event
	title = tsin_exhibition.10.title
	desc = tsin_exhibition.10.desc
	illustration_tags = {
		10 = exterior
		10 = happy
	}

	option = {
		name = tsin_exhibition.10.a
		add_prestige = prestige_severe_bonus
	}
}
