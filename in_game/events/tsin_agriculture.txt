# The guano boom
tsin_agriculture.10 = {
	type = country_event
	title = tsin_agriculture.10.title
	desc = tsin_agriculture.10.desc
	illustration_tags = {
		10 = exterior
		10 = regular
	}

	trigger = {
		tsin_flavor_budget_available_trigger = yes
		has_advance = tsin_agricultural_chemistry_advance
		NOT = { has_variable = tsin_agriculture_guano_done }
	}

	immediate = {
		tsin_spend_flavor_budget_effect = yes
		set_variable = tsin_agriculture_guano_done
	}

	# Buy guano by the shipload
	option = {
		name = tsin_agriculture.10.a
		change_gold_effect = { scale = -2 }
		add_country_modifier = { modifier = tsin_agriculture_guano_imports years = modifier_duration_years_normal mode = add_and_extend }
		ai_chance = {
			factor = 10
			modifier = {
				factor = 0
				gold < tsin_gold_cost_2
			}
		}
	}

	# Look for saltpeter at home
	option = {
		name = tsin_agriculture.10.b
		add_country_modifier = { modifier = tsin_agriculture_home_saltpeter years = modifier_duration_years_normal mode = add_and_extend }
		ai_chance = {
			factor = 5
		}
	}
}

# Grain Invasion: the farm question (ends the crisis for this country)
tsin_agriculture.81 = {
	type = country_event
	title = tsin_agriculture.81.title
	desc = tsin_agriculture.81.desc
	illustration_tags = {
		10 = interior
		10 = regular
	}

	trigger = {
		has_country_modifier = tsin_grain_invasion_crisis
	}

	# Protective grain tariffs
	option = {
		name = tsin_agriculture.81.a
		remove_country_modifier = tsin_grain_invasion_crisis
		add_country_modifier = { modifier = tsin_grain_tariffs years = modifier_duration_years_long mode = add_and_extend }
		change_societal_value = {
			type = mercantilism_vs_free_trade
			value = societal_value_minor_move_to_left
		}
		add_estate_satisfaction = { type = estate_type:nobles_estate value = estate_satisfaction_mild_bonus }
		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_weak_penalty }
		ai_chance = {
			factor = 5
			modifier = {
				factor = 2
				is_situation_active = situation:tsin_long_depression
			}
		}
	}

	# Cheap bread for the cities
	option = {
		name = tsin_agriculture.81.b
		remove_country_modifier = tsin_grain_invasion_crisis
		add_country_modifier = { modifier = tsin_cheap_bread years = modifier_duration_years_long mode = add_and_extend }
		change_societal_value = {
			type = mercantilism_vs_free_trade
			value = societal_value_minor_move_to_right
		}
		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_mild_bonus }
		add_estate_satisfaction = { type = estate_type:peasants_estate value = estate_satisfaction_mild_penalty }
		ai_chance = {
			factor = 5
			modifier = {
				factor = 2
				has_variable = tsin_factory_share
				var:tsin_factory_share >= 0.1
			}
		}
	}

	# Modernize the farms
	option = {
		name = tsin_agriculture.81.c
		trigger = {
			has_advance = tsin_steam_threshing_advance
		}
		remove_country_modifier = tsin_grain_invasion_crisis
		change_gold_effect = { scale = -3 }
		add_country_modifier = { modifier = tsin_farm_modernization years = modifier_duration_years_long mode = add_and_extend }
		ai_chance = {
			factor = 5
			modifier = {
				factor = 0
				gold < tsin_gold_cost_3
			}
		}
	}
}
