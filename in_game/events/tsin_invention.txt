namespace = tsin_invention

# Invention flavor (M3). Each fires once per country, only after the matching advance is researched,
# and asks where to invest the new technique. Fired from tsin_invention_yearly_pulse; spends the flavor budget.

# The Bessemer converter
tsin_invention.10 = {
	type = country_event
	title = tsin_invention.10.title
	desc = tsin_invention.10.desc
	illustration_tags = {
		10 = interior
		10 = regular
	}

	trigger = {
		tsin_flavor_budget_available_trigger = yes
		has_advance = tsin_bessemer_process_advance
		NOT = { has_variable = tsin_invention_bessemer_done }
	}

	immediate = {
		tsin_spend_flavor_budget_effect = yes
		set_variable = tsin_invention_bessemer_done
	}

	# License the process to private ironmasters
	option = {
		name = tsin_invention.10.a
		add_country_modifier = { modifier = tsin_invention_bessemer_licenses years = modifier_duration_years_long mode = add_and_extend }
		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_weak_bonus }
		ai_chance = {
			factor = 10
		}
	}

	# Build state converters for rails and guns
	option = {
		name = tsin_invention.10.b
		change_gold_effect = { scale = -3 }
		add_country_modifier = { modifier = tsin_invention_bessemer_state_works years = modifier_duration_years_long mode = add_and_extend }
		ai_chance = {
			factor = 5
			modifier = {
				factor = 0
				gold < tsin_gold_cost_3
			}
		}
	}
}
