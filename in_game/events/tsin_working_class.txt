# The factory clock: the foremen set the clocks back to steal minutes (needs standard time)
tsin_working_class.10 = {
	type = country_event
	title = tsin_working_class.10.title
	desc = tsin_working_class.10.desc
	outcome = neutral
	illustration_tags = {
		10 = interior
		10 = angry
	}

	trigger = {
		tsin_flavor_budget_available_trigger = yes
		has_variable = tsin_standard_time
		tsin_has_factory_trigger = yes
		NOT = { has_variable = tsin_factory_clock_done }
	}

	immediate = {
		tsin_spend_flavor_budget_effect = yes
		set_variable = tsin_factory_clock_done
	}

	# Seal the clocks and post inspectors
	option = {
		name = tsin_working_class.10.a
		add_country_modifier = { modifier = tsin_sealed_factory_clocks years = 20 }
		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_mild_penalty }
		ai_chance = {
			factor = 5
		}
	}

	# Time is money
	option = {
		name = tsin_working_class.10.b
		add_country_modifier = { modifier = tsin_stolen_minutes years = 10 }
		tsin_change_labor_grievance_effect = { VALUE = 5 }
		ai_chance = {
			factor = 5
		}
	}
}
