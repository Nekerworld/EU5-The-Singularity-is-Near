namespace = tsin_frontier

# Gold rushes (0.7, docs/frontier.md): discovery (.10), the boomtown (.20), gold inflation (.30).
# Fired from tsin_frontier_yearly_pulse; no flavor budget. Location variable tsin_gold_rush marks a rush (10 years).

# Gold in the creek
tsin_frontier.10 = {
	type = country_event
	title = tsin_frontier.10.title
	desc = tsin_frontier.10.desc
	illustration_tags = {
		10 = exterior
		10 = happy
	}

	trigger = {
		NOT = { has_variable = tsin_gold_rush_cooldown }
		any_owned_location = { tsin_gold_rush_field_trigger = yes }
	}

	immediate = {
		set_variable = { name = tsin_gold_rush_cooldown value = yes years = 15 }
		random_owned_location = {
			limit = { tsin_gold_rush_field_trigger = yes }
			save_scope_as = tsin_gold_field
			set_variable = { name = tsin_gold_rush value = yes years = 10 }
		}
	}

	# Let the prospectors come
	option = {
		name = tsin_frontier.10.a
		scope:tsin_gold_field = {
			add_location_modifier = { modifier = tsin_gold_rush_boom years = 10 mode = add_and_extend }
		}
		set_variable = { name = tsin_gold_rush_active value = yes years = 10 }
		ai_chance = {
			factor = 10
		}
	}

	# Licence the diggings to the crown
	option = {
		name = tsin_frontier.10.b
		scope:tsin_gold_field = {
			add_location_modifier = { modifier = tsin_gold_rush_licensed years = 10 mode = add_and_extend }
		}
		change_gold_effect = { scale = 2 }
		ai_chance = {
			factor = 5
		}
	}
}

# The boomtown
tsin_frontier.20 = {
	type = country_event
	title = tsin_frontier.20.title
	desc = tsin_frontier.20.desc
	illustration_tags = {
		10 = exterior
		10 = regular
	}

	trigger = {
		has_variable = tsin_gold_rush_active
		any_owned_location = {
			has_variable = tsin_gold_rush
			NOT = { has_variable = tsin_boomtown_done }
		}
	}

	immediate = {
		random_owned_location = {
			limit = {
				has_variable = tsin_gold_rush
				NOT = { has_variable = tsin_boomtown_done }
			}
			save_scope_as = tsin_gold_field
			set_variable = { name = tsin_boomtown_done value = yes years = 20 }
		}
	}

	# Incorporate the town: courts, sheriffs, a mint
	option = {
		name = tsin_frontier.20.a
		change_gold_effect = { scale = -1 }
		scope:tsin_gold_field = {
			add_location_modifier = { modifier = tsin_boomtown years = 20 mode = add_and_extend }
		}
		ai_chance = {
			factor = 10
		}
	}

	# Let the miners keep their own law
	option = {
		name = tsin_frontier.20.b
		add_estate_satisfaction = { type = estate_type:peasants_estate value = estate_satisfaction_weak_bonus }
		ai_chance = {
			factor = 3
		}
	}
}

# Gold floods the market
tsin_frontier.30 = {
	type = country_event
	title = tsin_frontier.30.title
	desc = tsin_frontier.30.desc
	illustration_tags = {
		10 = interior
		10 = regular
	}

	trigger = {
		has_variable = tsin_gold_rush_active
		NOT = { has_variable = tsin_gold_inflation_done }
	}

	immediate = {
		set_variable = { name = tsin_gold_inflation_done value = yes years = 10 }
	}

	# Mint it all
	option = {
		name = tsin_frontier.30.a
		change_gold_effect = { scale = 3 }
		add_country_modifier = { modifier = tsin_gold_inflation years = 5 }
		ai_chance = {
			factor = 10
		}
	}

	# Keep it in the vaults as reserves
	option = {
		name = tsin_frontier.30.b
		if = {
			limit = { NOT = { has_variable = tsin_gold_reserves } }
			set_variable = { name = tsin_gold_reserves value = 0 }
		}
		change_variable = { name = tsin_gold_reserves add = 1 }
		add_prestige = prestige_weak_bonus
		custom_tooltip = tsin_frontier.30.b.tt
		ai_chance = {
			factor = 5
		}
	}
}
