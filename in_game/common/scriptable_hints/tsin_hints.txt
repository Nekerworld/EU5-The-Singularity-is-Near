# Mechanic hints. Evaluated often, so only cheap country-level triggers.
hint_tsin_electric_power = {
	priority = {
		has_advance = tsin_dynamo_advance
		total_effective_building_levels:tsin_power_station < 1
	}
	hide = {
		NOT = { has_advance = tsin_dynamo_advance }
	}
	sort_priority = 50
	hint_tag = tsin_electric_power_concept
}
