# Time compression and population caps (docs/tempo.md).

# World population target as a multiple of the 1837 base: 1837 x1.0, 1900 x1.6, 1950 x2.4, 2000 x5.5, 2100+ x7.0.
tsin_pop_target_multiplier = {
	value = 1
	if = {
		limit = { current_year < 1900 }
		add = {
			value = current_year
			subtract = 1837
			multiply = 0.0095238	# 0.6 / 63
			min = 0
		}
	}
	else_if = {
		limit = { current_year < 1950 }
		add = 0.6
		add = {
			value = current_year
			subtract = 1900
			multiply = 0.016	# 0.8 / 50
		}
	}
	else_if = {
		limit = { current_year < 2000 }
		add = 1.4
		add = {
			value = current_year
			subtract = 1950
			multiply = 0.062	# 3.1 / 50
		}
	}
	else = {
		add = 4.5
		add = {
			value = current_year
			subtract = 2000
			multiply = 0.015	# 1.5 / 100
		}
	}
	max = tsin_pop_absolute_cap_multiplier
}

# Earth today is about 7x its 1837 population.
tsin_pop_absolute_cap_multiplier = 7

# World population over the target curve. 1 = on target.
tsin_world_pop_ratio = {
	value = 0
	if = {
		limit = {
			has_global_variable = tsin_world_pop_base
			global_var:tsin_world_pop_base > 0
		}
		add = tsin_obs_world_population
		divide = global_var:tsin_world_pop_base
		divide = tsin_pop_target_multiplier
	}
}

# World population over the 1837 base.
tsin_world_pop_growth_multiple = {
	value = 0
	if = {
		limit = {
			has_global_variable = tsin_world_pop_base
			global_var:tsin_world_pop_base > 0
		}
		add = tsin_obs_world_population
		divide = global_var:tsin_world_pop_base
	}
}

# root = country. Industrial acceleration score (tiers: 2 / 3 / 5).
tsin_acceleration_points = {
	value = 0
	if = {
		limit = { has_embraced_institution = institution:tsin_industrial_capitalism }
		add = 1
	}
	if = {
		limit = { has_embraced_institution = institution:tsin_mass_society }
		add = 1
	}
	if = {
		limit = { has_embraced_institution = institution:tsin_electrification }
		add = 1
	}
	if = {
		limit = {
			has_variable = tsin_factory_share
			var:tsin_factory_share >= 0.08
		}
		add = 1
	}
	if = {
		limit = {
			has_variable = tsin_factory_share
			var:tsin_factory_share >= 0.2
		}
		add = 1
	}
	if = {
		limit = { has_variable = tsin_standard_time }
		add = 1
	}
}

# Country brake threshold: yearly population growth above 3%.
tsin_population_pressure_threshold = 0.03

# root = country. Population growth since the last yearly pulse, as a fraction.
tsin_country_pop_growth_rate = {
	value = 0
	if = {
		limit = {
			has_variable = tsin_last_population
			var:tsin_last_population > 0
		}
		add = total_population
		subtract = var:tsin_last_population
		divide = var:tsin_last_population
	}
}

# root = country. Share of the population living in towns and cities (annual aggregate).
tsin_country_urban_share = {
	value = 0
	if = {
		limit = {
			has_variable = tsin_agg_urban_pop
			total_population > 0
		}
		add = var:tsin_agg_urban_pop
		divide = total_population
		max = 1
	}
}

# root = country. Demographic transition stage 0..2 from literacy and urbanization.
tsin_demographic_stage = {
	value = 0
	if = {
		limit = {
			average_country_literacy >= 50
			tsin_country_urban_share >= 0.25
		}
		add = 1
	}
	if = {
		limit = {
			average_country_literacy >= 75
			tsin_country_urban_share >= 0.4
		}
		add = 1
	}
}
