# Factory jobs in the location, in pop-size units (employment_size 1 = 1000 people).
tsin_location_industrial_jobs = {
	value = tsin_location_vanilla_mill_levels
	add = "location_building_level(building_type:tsin_machine_works)"
	add = "location_building_level(building_type:tsin_fertilizer_plant)"
	add = "location_building_level(building_type:tsin_explosives_plant)"
	add = "location_building_level(building_type:tsin_farm_machinery_works)"
	multiply = mills_employment
	add = {
		value = tsin_location_automated_mill_levels
		multiply = tsin_mechanized_mill_employment
	}
}

# Factory jobs as a share of the location population, 0..1.
tsin_location_industrial_share = {
	value = 0
	if = {
		limit = { population > 0 }
		add = tsin_location_industrial_jobs
		divide = population
		max = 1
	}
}

# 0..1 scale used by tsin_industrialization: saturates at this factory-job share.

# Share of factory levels that are automated upgrades, 0..1 (tsin_automation).

	value = 0
	if = {
		limit = { tsin_location_mill_levels > 0 }

		divide = tsin_location_mill_levels
		max = 1
	}
}

# Factory site quality, 0..1: market access, coal or iron, a railway station, city rank (tsin_industrial_siting).
tsin_location_siting_score = {
	value = market_access
	multiply = 0.4
	if = {
		limit = {
			OR = {
				raw_material = goods:coal
				raw_material = goods:iron
			}
		}
		add = 0.25
	}
	if = {
		limit = { has_building_with_at_least_one_level = tsin_railway_station }
		add = 0.2
	}
	if = {
		limit = {
			OR = {
				location_rank = location_rank:city
				location_rank = location_rank:megalopolis
			}
		}
		add = 0.15
	}
	max = 1
}

# Owner's per-capita GDP over the highest in the world last year, 0..1 (tsin_gdp_per_capita).
tsin_location_gdp_pc_factor = {
	value = 0
	if = {
		limit = {
			owner ?= { has_variable = tsin_gdp_per_capita }
			has_global_variable = tsin_world_gdp_pc_max
			global_var:tsin_world_gdp_pc_max > 0
		}
		value = owner.var:tsin_gdp_per_capita
		divide = global_var:tsin_world_gdp_pc_max
		max = 1
	}
}

# Monthly food balance per pop of the location's market (vanilla market_food_balance map mode).
tsin_location_market_food_per_pop = {
	value = 0
	if = {
		limit = { market ?= { market_population > 0 } }
		value = market.market_monthly_food_balance
		divide = market.market_population
	}
}
