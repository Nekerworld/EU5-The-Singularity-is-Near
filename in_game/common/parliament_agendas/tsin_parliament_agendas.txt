# Age VII parliament agendas (M3 corn laws). Mirrors vanilla pa_move_towards_free_trade / mercantilism.
# Exactly one of tsin_corn_laws_protection / tsin_corn_laws_repealed is active after the first vote.

pa_tsin_repeal_corn_laws = {
	estate = burghers_estate

	potential = {
		current_age_or_later = { age = age_7_industrialization }
		NOT = { has_country_modifier = tsin_corn_laws_repealed }
	}

	on_accept = {
		if = {
			limit = { has_country_modifier = tsin_corn_laws_protection }
			remove_country_modifier = tsin_corn_laws_protection
		}
		add_country_modifier = { modifier = tsin_corn_laws_repealed months = -1 }
		change_societal_value = {
			type = mercantilism_vs_free_trade
			value = societal_value_minor_move_to_right
		}
		trigger_event_non_silently = tsin_corn_laws.2
	}

	chance = {
		value = 10
		if = {
			limit = { societal_value:mercantilism_vs_free_trade > 0 }
			add = 10
		}
	}
}

# M4 abolition: the clergy push the vanilla slavery_outlawed policy once abolitionism is strong.
pa_tsin_abolish_slavery = {
	estate = clergy_estate

	potential = {
		current_age_or_later = { age = age_7_industrialization }
		NOT = { has_policy = policy:slavery_outlawed }
		has_variable = tsin_abolitionism
		var:tsin_abolitionism >= 50
	}

	on_accept = {
		tsin_table_bill_effect = { POLICY = policy:slavery_outlawed BONUS = 0.25 }
		change_societal_value = {
			type = spiritualist_vs_humanist
			value = societal_value_minor_move_to_right
		}
	}

	chance = {
		value = 5
		if = {
			limit = { var:tsin_abolitionism >= 75 }
			add = 15
		}
	}
}

# 0.7 labor (docs/labor.md): the commoners bring the labor grievance to parliament.
pa_tsin_factory_inspection = {
	estate = peasants_estate

	potential = {
		current_age_or_later = { age = age_7_industrialization }
		has_variable = tsin_labor_grievance
		var:tsin_labor_grievance >= 30
		NOT = { has_country_modifier = tsin_factory_inspection_agenda }
	}

	on_accept = {
		add_country_modifier = { modifier = tsin_factory_inspection_agenda years = 10 }
		change_variable = { name = tsin_labor_grievance add = -10 }
		clamp_variable = { name = tsin_labor_grievance min = 0 max = 100 }
	}

	chance = {
		value = 10
		if = {
			limit = { var:tsin_labor_grievance >= 50 }
			add = 10
		}
	}
}
