# Vanilla game_triggers.txt assumes age 6 is the last age. These replacements keep
# current_age_or_later = { age = age_5_absolutism / age_6_revolutions } true in age 7.
# Recorded in docs/overrides.md.

TRY_REPLACE:current_later_age_5_absolutism_trigger = {
	custom_tooltip = {
		text = current_age_at_least_5_tt
		OR = {
			current_age = age_5_absolutism
			current_age = age_6_revolutions
			current_age = age_7_industrialization
		}
	}
}

TRY_REPLACE:current_later_age_6_revolutions_trigger = {
	custom_tooltip = {
		text = current_age_at_least_6_tt
		OR = {
			current_age = age_6_revolutions
			current_age = age_7_industrialization
		}
	}
}

current_later_age_7_industrialization_trigger = {
	custom_tooltip = {
		text = current_age_at_least_7_tt
		current_age = age_7_industrialization
	}
}

# Age 7 specialization choices (tsin_4_choices.txt): the specialization picked in ages_of_eu.1 opens
# its ten choices at once; the others open when the paired institution is embraced (docs/tech_tree_age7.md).
tsin_adm_specialization_open_trigger = {
	OR = {
		age_preference = adm
		has_embraced_institution = institution:tsin_mass_society
	}
}

tsin_dip_specialization_open_trigger = {
	OR = {
		age_preference = dip
		has_embraced_institution = institution:tsin_industrial_capitalism
	}
}

tsin_mil_specialization_open_trigger = {
	OR = {
		age_preference = mil
		has_embraced_institution = institution:tsin_electrification
	}
}

# Short helpers for mod content.
tsin_is_age_7_trigger = {
	current_age = age_7_industrialization
}
