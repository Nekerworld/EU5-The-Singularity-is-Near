# Yearly update of the age VII movement variables (script_values/tsin_movement_values.txt). root = country.
# Called from tsin_yearly_country_pulse.

tsin_update_movements_effect = {
	if = {
		limit = { has_slavery = yes }
		if = {
			limit = { NOT = { has_variable = tsin_abolitionism } }
			set_variable = { name = tsin_abolitionism value = 0 }
		}
		change_variable = { name = tsin_abolitionism add = tsin_abolitionism_yearly_growth }
		clamp_variable = { name = tsin_abolitionism min = 0 max = tsin_movement_max }
	}
	tsin_movement_tick_effect = { VAR = tsin_liberalism GROWTH = tsin_liberalism_yearly_growth AGITATION = tsin_liberalism_agitation }
	tsin_movement_tick_effect = { VAR = tsin_nationalism GROWTH = tsin_nationalism_yearly_growth AGITATION = tsin_nationalism_agitation }
	tsin_movement_tick_effect = { VAR = tsin_labor_unionism GROWTH = tsin_labor_unionism_yearly_growth AGITATION = tsin_labor_unionism_agitation }
	tsin_movement_tick_effect = { VAR = tsin_socialism GROWTH = tsin_socialism_yearly_growth AGITATION = tsin_socialism_agitation }
	tsin_movement_tick_effect = { VAR = tsin_womens_suffrage GROWTH = tsin_womens_suffrage_yearly_growth AGITATION = tsin_womens_suffrage_agitation }

	# Strongest movement, read by the tsin_radicalism map mode
	set_variable = { name = tsin_radicalism value = 0 }
	tsin_radicalism_consider_effect = { VAR = tsin_abolitionism }
	tsin_radicalism_consider_effect = { VAR = tsin_liberalism }
	tsin_radicalism_consider_effect = { VAR = tsin_nationalism }
	tsin_radicalism_consider_effect = { VAR = tsin_labor_unionism }
	tsin_radicalism_consider_effect = { VAR = tsin_socialism }
	tsin_radicalism_consider_effect = { VAR = tsin_womens_suffrage }
}

tsin_radicalism_consider_effect = {
	if = {
		limit = {
			has_variable = $VAR$
			var:$VAR$ > var:tsin_radicalism
		}
		set_variable = { name = tsin_radicalism value = var:$VAR$ }
	}
}

# One movement: grow, clamp to 0..tsin_movement_max, toggle the agitation modifier with hysteresis.
tsin_movement_tick_effect = {
	if = {
		limit = { NOT = { has_variable = $VAR$ } }
		set_variable = { name = $VAR$ value = 0 }
	}
	change_variable = { name = $VAR$ add = $GROWTH$ }
	clamp_variable = { name = $VAR$ min = 0 max = tsin_movement_max }
	if = {
		limit = {
			NOT = { has_country_modifier = $AGITATION$ }
			var:$VAR$ >= tsin_movement_agitation_start
		}
		add_country_modifier = { modifier = $AGITATION$ months = -1 }
	}
	else_if = {
		limit = {
			has_country_modifier = $AGITATION$
			var:$VAR$ < tsin_movement_agitation_end
		}
		remove_country_modifier = $AGITATION$
	}
}
# Frees every slave pop of root (pattern of the vanilla slaves rebel demand).
tsin_emancipate_all_slaves_effect = {
	every_pop = {
		limit = {
			owner = root
			pop_type = pop_type:slaves
		}
		change_pop_type = pop_type:peasants
	}
}
