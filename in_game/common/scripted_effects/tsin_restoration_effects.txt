tsin_restore_russia_effect = {
# Full guard must pass before creating a country, moving a city, or transferring gold.
if = { limit = { tsin_restore_russia_trigger = yes }
if = { limit = { NOT = { has_global_variable = tsin_restore_russia_count } } set_global_variable = { name = tsin_restore_russia_count value = 0 } }
change_global_variable = { name = tsin_restore_russia_count add = 1 }
set_global_variable = { name = tsin_restore_russia_next value = tsin_restoration_next_year }
location:moscow = {
save_scope_as = tsin_restore_russia_centre
owner = { save_scope_as = tsin_restore_russia_sponsor }
create_country_from_location = {
save_scope_as = tsin_restore_russia_country
set_capital = scope:tsin_restore_russia_centre
change_culture = scope:tsin_restore_russia_centre.dominant_culture
change_religion = scope:tsin_restore_russia_centre.dominant_religion
change_government_type = government_type:monarchy
# Do not inherit an unfinanced treasury endowment from engine defaults.
add_gold = { value = gold multiply = -1 }
}
}

location:moscow = { if = { limit = { NOT = { owner ?= scope:tsin_restore_russia_country } } change_location_owner = scope:tsin_restore_russia_country change_location_controller = scope:tsin_restore_russia_country } }
location:novgorod = { if = { limit = { NOT = { owner ?= scope:tsin_restore_russia_country } } change_location_owner = scope:tsin_restore_russia_country change_location_controller = scope:tsin_restore_russia_country } }
location:tver = { if = { limit = { NOT = { owner ?= scope:tsin_restore_russia_country } } change_location_owner = scope:tsin_restore_russia_country change_location_controller = scope:tsin_restore_russia_country } }
location:vladimir = { if = { limit = { NOT = { owner ?= scope:tsin_restore_russia_country } } change_location_owner = scope:tsin_restore_russia_country change_location_controller = scope:tsin_restore_russia_country } }
location:moscow.province_definition = { every_location_in_province_definition = {
limit = { owner ?= scope:tsin_restore_russia_sponsor controller = owner is_capital = no is_historical_capital = no
owner = { tsin_history_ai_realm = yes at_war = no num_locations > 1 } }
change_location_owner = scope:tsin_restore_russia_country change_location_controller = scope:tsin_restore_russia_country
} }
location:novgorod.province_definition = { every_location_in_province_definition = {
