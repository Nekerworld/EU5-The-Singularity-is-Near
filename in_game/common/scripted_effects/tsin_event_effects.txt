# The Singularity is Near - event framework effects. Scope: country.

# Every tsin_ flavor event calls this in its immediate block.
tsin_spend_flavor_budget_effect = {
	if = {
		limit = { has_variable = tsin_flavor_events_this_year }
		change_variable = { name = tsin_flavor_events_this_year add = 1 }
	}
	else = {
		set_variable = { name = tsin_flavor_events_this_year value = 1 }
	}
	if = {
		limit = { tsin_observer_logging_trigger = yes }
		if = {
			limit = { has_global_variable = tsin_obs_flavor_events }
			change_global_variable = { name = tsin_obs_flavor_events add = 1 }
		}
		else = {
			set_global_variable = { name = tsin_obs_flavor_events value = 1 }
		}
	}
}

tsin_reset_flavor_budget_effect = {
	if = {
		limit = { has_variable = tsin_flavor_events_this_year }
		remove_variable = tsin_flavor_events_this_year
	}
}

# Counts ports opened under foreign pressure (treaty port peace treaty, tsin_flavor.10). Opens tsin_modernization.
tsin_add_opening_pressure_effect = {
	if = {
		limit = { has_variable = tsin_treaty_ports_opened }
		change_variable = { name = tsin_treaty_ports_opened add = 1 }
	}
	else = {
		set_variable = { name = tsin_treaty_ports_opened value = 1 }
	}
}

tsin_flavor_end_serfdom_effect = {
	if = {
		limit = { has_estate_privilege = estate_privilege:noble_serfdom_rights }
		revoke_estate_privilege = estate_privilege:noble_serfdom_rights
	}
}
