# The Singularity is Near - on_action hooks.
# Vanilla hooks are extended by merging an on_actions list; vanilla bodies are left untouched.

on_new_age = {
	on_actions = { tsin_on_new_age }
}

on_new_age_global = {
	on_actions = { tsin_on_new_age_global }
}

yearly_country_pulse = {
	on_actions = {
		tsin_observer_yearly_pulse
		tsin_yearly_country_pulse
		tsin_economy_yearly_pulse
		tsin_invention_yearly_pulse
		tsin_military_yearly_pulse
		tsin_art_yearly_pulse
		tsin_flavor_yearly_pulse
		tsin_society_yearly_pulse
		tsin_hunger_yearly_pulse
		tsin_society_flavor_pulse
	}
}

on_institution_embraced = {
	on_actions = { tsin_on_institution_embraced }
}

# root = country
tsin_on_new_age = {
	trigger = {
		current_age = age_7_industrialization
	}
	events = {
		tsin_system.2
	}
}

tsin_on_new_age_global = {
	effect = {
		if = {
			limit = {
				current_age = age_7_industrialization
				NOT = { has_global_variable = tsin_version }
			}
			set_global_variable = { name = tsin_version value = 1 }
			set_global_variable = { name = tsin_age_7_started value = 1 }
		}
	}
}

# root = country. Also migrates saves that entered age 7 before the mod was enabled.
tsin_yearly_country_pulse = {
	trigger = {
		current_age_or_later = { age = age_7_industrialization }
	}
	effect = {
		tsin_reset_flavor_budget_effect = yes
		tsin_update_movements_effect = yes
	}
	events = {
		tsin_system.1
		tsin_system.10
		tsin_system.11
		tsin_system.12
	}
}

# root = country. Observer-run metrics: the first country to pulse each ~year writes one log line (docs/tests/observer_run.md).
tsin_observer_yearly_pulse = {
	trigger = {
		tsin_observer_logging_trigger = yes
		NOT = { has_global_variable = tsin_obs_logged_recently }
	}
	effect = {
		set_global_variable = { name = tsin_obs_logged_recently value = yes days = 360 }
		debug_log = tsin_observer_log_line
		if = {
			limit = { has_global_variable = tsin_obs_flavor_events }
			remove_global_variable = tsin_obs_flavor_events
		}
	}
}

# root = country. Factory flavor events; the budget trigger caps them per year.
tsin_economy_yearly_pulse = {
	trigger = {
		current_age_or_later = { age = age_7_industrialization }
		tsin_flavor_budget_available_trigger = yes
	}
	random_events = {
		10 = tsin_economy.10
		10 = tsin_economy.20
		10 = tsin_economy.30
		10 = tsin_economy.40
		5 = tsin_corn_laws.1
		5 = tsin_corn_laws.3
		10 = tsin_economy.50
		10 = tsin_economy.60
		40 = tsin_economy.61
		10 = tsin_economy.70
		10 = tsin_economy.80

		chance_to_happen = 25
	}
}

# root = country, scope:target = institution
tsin_on_institution_embraced = {
	effect = {
		if = {
			limit = {
				scope:target = institution:tsin_industrial_capitalism
				NOT = { has_global_variable = tsin_first_embraced_industrial_capitalism }
			}
			set_global_variable = { name = tsin_first_embraced_industrial_capitalism value = root }
		}
		else_if = {
			limit = {
				scope:target = institution:tsin_mass_society
				NOT = { has_global_variable = tsin_first_embraced_mass_society }
			}
			set_global_variable = { name = tsin_first_embraced_mass_society value = root }
		}
		else_if = {
			limit = {
				scope:target = institution:tsin_electrification
				NOT = { has_global_variable = tsin_first_embraced_electrification }
			}
			set_global_variable = { name = tsin_first_embraced_electrification value = root }
		}
	}
}
