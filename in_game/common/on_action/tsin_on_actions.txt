tsin_on_new_age_global = {
	effect = {
		if = {
			limit = {
				current_age = age_7_industrialization
				NOT = { has_global_variable = tsin_version }
			}
			set_global_variable = { name = tsin_version value = 3 }
			set_global_variable = { name = tsin_age_7_started value = 1 }
			tsin_init_world_pop_base_effect = yes
		}
	}
}

# root = country. Observer-run metrics: the first country to pulse each ~year writes one log line (docs/tests/observer_run.md).
tsin_observer_yearly_pulse = {
	trigger = {
		tsin_observer_logging_trigger = yes
		NOT = { has_global_variable = tsin_obs_logged_recently }
	}
	effect = {
		set_global_variable = { name = tsin_obs_logged_recently value = yes days = 360 }
		debug_log = tsin_observer_log_line
		if = {
			limit = { has_global_variable = tsin_obs_flavor_events }
			remove_global_variable = tsin_obs_flavor_events
		}
	}
}

# root = country. Bill readings every 3 months (docs/legislation.md). Only countries with a bill pay for it.
tsin_bill_monthly_pulse = {
	trigger = {
		has_variable = tsin_bill_policy
		NOT = { has_variable = tsin_bill_check_timer }
	}
	effect = {
		if = {
			limit = { tsin_bill_system_active_trigger = yes }
			tsin_bill_reading_effect = yes
		}
		else = {
			tsin_clear_bill_effect = yes
		}
	}
}

# Legacy entry point retained for compatibility; the shared capital pulse now owns the stock clock.
tsin_stock_monthly_pulse = {
	trigger = {
		current_age_or_later = { age = age_7_industrialization }
		NOT = { has_variable = tsin_stock_timer }
		OR = {
			has_variable = tsin_has_stock_market
			tsin_company_count > 0
		}
	}
	effect = {
		set_variable = { name = tsin_stock_timer months = 3 }
		if = {
			limit = { NOT = { has_variable = tsin_has_stock_market } }
			tsin_init_existing_companies_effect = yes
		}
		else_if = {
			limit = { tsin_company_count > 0 }
			tsin_update_stocks_effect = yes
			if = {
				limit = { tsin_has_chairman_trigger = no }
				trigger_event_non_silently = tsin_corporate.10
			}
		}
		else = {
			tsin_dissolve_all_companies_effect = yes
		}
	}
}

# root = country, scope:target = institution
tsin_on_institution_embraced = {
	effect = {
		if = {
			limit = {
				scope:target = institution:tsin_industrial_capitalism
				NOT = { has_global_variable = tsin_first_embraced_industrial_capitalism }
			}
			set_global_variable = { name = tsin_first_embraced_industrial_capitalism value = root }
		}
		else_if = {
			limit = {
				scope:target = institution:tsin_mass_society
				NOT = { has_global_variable = tsin_first_embraced_mass_society }
			}
			set_global_variable = { name = tsin_first_embraced_mass_society value = root }
		}
		else_if = {
			limit = {
				scope:target = institution:tsin_electrification
				NOT = { has_global_variable = tsin_first_embraced_electrification }
			}
			set_global_variable = { name = tsin_first_embraced_electrification value = root }
		}
	}
}
