# Age VII chemical industry: fertilizer and explosives, and the rural buildings that consume them.
# Mill inputs are priced so that output value is about input cost * (1 + mills_profit_margin), as in vanilla mills.
# Prices: saltpeter 2, alum 3, coal 2, fish 1, livestock 1.5, lumber 1.5, tsin_fertilizer 2.5, tsin_explosives 5.
# The rural enhancement buildings follow vanilla windmill/local_smelters: a maintenance-only method plus a modifier.
# Whether the modifier weakens when fertilizer or explosives are short is unverified (docs/economy_chain.md).

	unique_production_methods = {

		tsin_fertilizer_mineral = {

		}
